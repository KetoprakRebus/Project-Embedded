#include "DHT.h"

// PIN KOMPONEN ARDUINO UNO
#define SOIL_PIN A0       // Soil moisture AO ke A0
#define DHT_PIN 4         // DATA DHT11/DHT22 ke D4
#define RELAY_PIN 7       // IN relay ke D7

// TIPE DHT
// #define DHTTYPE DHT11
#define DHTTYPE DHT22

DHT dht(DHT_PIN, DHTTYPE);

// RELAY
// Relay aktif LOW
#define RELAY_ON LOW
#define RELAY_OFF HIGH

// BATAS SENSOR ARDUINO UNO
// Nilai soil Arduino Uno: 0 - 1023
// Nilai besar = tanah kering
// Nilai kecil = tanah basah
int batasTanahKering = 600;
int batasTanahBasah = 400;

// Batas suhu dan kelembapan udara
float batasSuhuPanas = 32.0;
float batasKelembapanUdaraRendah = 60.0;

bool pompaNyala = false;

void setup() {
  Serial.begin(9600);

  dht.begin();

  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, RELAY_OFF);

  Serial.println("Sistem Penyiraman Otomatis Tanaman Cabai");
  Serial.println("Arduino Uno + Soil Moisture + DHT22 + Relay + Pompa");
}

void loop() {
  int nilaiTanah = analogRead(SOIL_PIN);

  float suhu = dht.readTemperature();
  float kelembapanUdara = dht.readHumidity();

  Serial.println("==================================");
  Serial.print("Nilai tanah: ");
  Serial.println(nilaiTanah);

  bool dhtTerbaca = true;

  if (isnan(suhu) || isnan(kelembapanUdara)) {
    Serial.println("DHT gagal terbaca");
    dhtTerbaca = false;
  } else {
    Serial.print("Suhu udara: ");
    Serial.print(suhu);
    Serial.println(" C");

    Serial.print("Kelembapan udara: ");
    Serial.print(kelembapanUdara);
    Serial.println(" %");
  }

  // KONDISI SENSOR TANAH
  bool tanahKering = nilaiTanah > batasTanahKering;
  bool tanahBasah = nilaiTanah < batasTanahBasah;
  bool tanahNormal = nilaiTanah >= batasTanahBasah && nilaiTanah <= batasTanahKering;

  // KONDISI UDARA
  bool udaraPanasDanKering = false;

  if (dhtTerbaca) {
    udaraPanasDanKering = suhu > batasSuhuPanas && kelembapanUdara < batasKelembapanUdaraRendah;
  }

  // LOGIKA POMPA
  // Prioritas 1: kalau tanah sudah basah, pompa wajib mati
  if (tanahBasah) {
    digitalWrite(RELAY_PIN, RELAY_OFF);
    pompaNyala = false;

    Serial.println("Status: TANAH BASAH");
    Serial.println("Pompa: MATI");
  }

  // Prioritas 2: kalau tanah kering atau udara panas dan kering, pompa menyala
  else if ((tanahKering || udaraPanasDanKering) && pompaNyala == false) {
    digitalWrite(RELAY_PIN, RELAY_ON);
    pompaNyala = true;

    Serial.println("Status: BUTUH PENYIRAMAN");
    Serial.println("Pompa: MENYALA");

    if (tanahKering) {
      Serial.println("Alasan: Tanah kering");
    }

    if (udaraPanasDanKering) {
      Serial.println("Alasan: Suhu panas dan kelembapan udara rendah");
    }
  }

  // Prioritas 3: kalau pompa sudah menyala, biarkan menyala sampai tanah basah
  else if (pompaNyala == true) {
    Serial.println("Status: PENYIRAMAN BERJALAN");
    Serial.println("Pompa: MASIH MENYALA");

    if (tanahNormal) {
      Serial.println("Keterangan: Tanah belum cukup basah untuk mematikan pompa");
    }
  }

  // Prioritas 4: kondisi tanah cukup lembap, pompa mati
  else {
    Serial.println("Status: NORMAL / CUKUP LEMBAP");
    Serial.println("Pompa: MATI");
  }

  delay(2000);
}
